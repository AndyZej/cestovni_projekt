@login_required def diskuze_zeme(request, slug): vybrana_zeme = get_object_or_404(Zeme, slug=slug) if request.method == 'POST': text_zpravy = request.POST.get('text') if text_zpravy: KomunitniZprava.objects.create( zeme=vybrana_zeme, uzivatel=request.user, text=text_zpravy ) return redirect('diskuze_zeme', slug=slug) zpravy = KomunitniZprava.objects.filter(zeme=vybrana_zeme).order_by('-cas_odeslani') # TADY JE ZMĚNA: Automaticky zkusí najít šablonu podle slugu (např. 'zpravy/komunita/zeme/kapverdy.html') # Pokud neexistuje, použije výchozí 'diskuze_zeme.html' sablona = f'zpravy/komunita/zeme/{slug}.html' return render(request, sablona, { 'zeme': vybrana_zeme, 'zpravy': zpravy })
